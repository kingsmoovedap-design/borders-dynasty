### `omega/omega.js`

```js
// Omega dashboard logic

const cfg = window.DYNASTY_CONFIG || {};
const STORAGE_KEY = "omega_auth";

function isAuthenticated() {
  return localStorage.getItem(STORAGE_KEY) === "true";
}

if (!isAuthenticated()) {
  window.location.href = "login.html";
}

const logoutBtn = document.getElementById('logoutBtn');
if (logoutBtn) {
  logoutBtn.addEventListener('click', () => {
    localStorage.removeItem(STORAGE_KEY);
    window.location.href = "login.html";
  });
}

const coinData = document.getElementById('coinData');
const escrowData = document.getElementById('escrowData');
const apiStatus = document.getElementById('apiStatus');
const pingApiBtn = document.getElementById('pingApiBtn');

if (coinData) {
  if (cfg.sovereignCoinAddress) {
    coinData.innerHTML = `
      Sovereign Coin: <a href="https://bscscan.com/token/${cfg.sovereignCoinAddress}" target="_blank">
      ${cfg.sovereignCoinAddress}</a>
    `;
  } else {
    coinData.textContent = "Sovereign Coin address not set in config.js.";
  }
}

if (escrowData) {
  if (cfg.escrowContractAddress) {
    escrowData.textContent = `Escrow Contract: ${cfg.escrowContractAddress}`;
  } else {
    escrowData.textContent = "Escrow contract address not set in config.js.";
  }
}

if (pingApiBtn && apiStatus) {
  pingApiBtn.addEventListener('click', async () => {
    if (!cfg.logisticsApiBaseUrl) {
      apiStatus.textContent = "Logistics API base URL not set in config.js.";
      return;
    }
    apiStatus.textContent = "Pinging API…";
    try {
      const res = await fetch(cfg.logisticsApiBaseUrl + "/health");
      if (res.ok) {
        const data = await res.json().catch(() => ({}));
        apiStatus.textContent = "API OK: " + (data.status || "healthy");
      } else {
        apiStatus.textContent = "API error: " + res.status;
      }
    } catch (e) {
      apiStatus.textContent = "API unreachable.";
      console.error(e);
    }
  });
}
```

### `loadboard/loadboard.js`

```js
const loadsGrid = document.getElementById('loadsGrid');
const loadsStatus = document.getElementById('loadsStatus');
const cfgLB = window.DYNASTY_CONFIG || {};

async function loadLoads() {
  if (!cfgLB.logisticsApiBaseUrl) {
    loadsStatus.textContent = "Logistics API base URL not set in config.js.";
    return;
  }
  loadsStatus.textContent = "Loading loads…";
  try {
    const res = await fetch(cfgLB.logisticsApiBaseUrl + "/loads");
    if (!res.ok) {
      loadsStatus.textContent = "Error loading loads: " + res.status;
      return;
    }
    const loads = await res.json();
    loadsGrid.innerHTML = '';
    if (!Array.isArray(loads) || loads.length === 0) {
      loadsStatus.textContent = "No loads available.";
      return;
    }
    loads.forEach(load => {
      const card = document.createElement('div');
      card.className = 'lb-card';
      card.innerHTML = `
        <h3>${load.reference || 'Load'}</h3>
        <p><strong>Origin:</strong> ${load.origin}</p>
        <p><strong>Destination:</strong> ${load.destination}</p>
        <p><strong>Rate:</strong> ${load.rate}</p>
        <p><strong>Status:</strong> ${load.status}</p>
      `;
      loadsGrid.appendChild(card);
    });
    loadsStatus.textContent = "";
  } catch (e) {
    console.error(e);
    loadsStatus.textContent = "Failed to load loads.";
  }
}

loadLoads();
```

### `driver-app/driver.js`

```js
const driverLoads = document.getElementById('driverLoads');
const driverStatus = document.getElementById('driverStatus');
const cfgDR = window.DYNASTY_CONFIG || {};

// For now, we assume a single driver ID hardcoded or later from login
const DRIVER_ID = "driver-001";

async function loadDriverLoads() {
  if (!cfgDR.logisticsApiBaseUrl) {
    driverStatus.textContent = "Logistics API base URL not set in config.js.";
    return;
  }
  driverStatus.textContent = "Loading your loads…";
  try {
    const res = await fetch(cfgDR.logisticsApiBaseUrl + "/drivers/" + DRIVER_ID + "/loads");
    if (!res.ok) {
      driverStatus.textContent = "Error loading loads: " + res.status;
      return;
    }
    const loads = await res.json();
    driverLoads.innerHTML = '';
    if (!Array.isArray(loads) || loads.length === 0) {
      driverStatus.textContent = "No assigned loads.";
      return;
    }
    loads.forEach(load => {
      const card = document.createElement('div');
      card.className = 'dr-card';
      card.innerHTML = `
        <h3>${load.reference || 'Load'}</h3>
        <p><strong>Origin:</strong> ${load.origin}</p>
        <p><strong>Destination:</strong> ${load.destination}</p>
        <p><strong>Status:</strong> ${load.status}</p>
      `;
      const btn = document.createElement('button');
      btn.textContent = "Mark as Delivered";
      btn.addEventListener('click', () => updateStatus(load.id, "delivered"));
      card.appendChild(btn);
      driverLoads.appendChild(card);
    });
    driverStatus.textContent = "";
  } catch (e) {
    console.error(e);
    driverStatus.textContent = "Failed to load loads.";
  }
}

async function updateStatus(loadId, status) {
  if (!cfgDR.logisticsApiBaseUrl) return;
  try {
    const res = await fetch(cfgDR.logisticsApiBaseUrl + "/loads/" + loadId + "/status", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ status, driverId: DRIVER_ID })
    });
    if (res.ok) {
      alert("Status updated.");
      loadDriverLoads();
    } else {
      alert("Failed to update status: " + res.status);
    }
  } catch (e) {
    console.error(e);
    alert("Error updating status.");
  }
}

loadDriverLoads();
```

```js
    if (!res.ok) {
      fleetStatus.textContent = "Error loading fleet: " + res.status;
      return;
    }
    const units = await res.json();
    fleetGrid.innerHTML = '';
    if (!Array.isArray(units) || units.length === 0) {
      fleetStatus.textContent = "No units available.";
      return;
    }
    units.forEach(unit => {
      const card = document.createElement('div');
      card.className = 'fl-card';
      card.innerHTML = `
        <h3>${unit.name}</h3>
        <p><strong>Type:</strong> ${unit.type}</p>
        <p><strong>Rate:</strong> ${unit.rate}</p>
        <p><strong>Status:</strong> ${unit.status}</p>
      `;
      fleetGrid.appendChild(card);
    });
    fleetStatus.textContent = "";
  } catch (e) {
    console.error(e);
    fleetStatus.textContent = "Failed to load fleet.";
  }
}

loadFleet();
```

### `logistics-api/server.js`

```js
const express = require('express');
const cors = require('cors');

const app = express();
app.use(cors());
app.use(express.json());

// Simple in-memory data for now
let loads = [
  { id: "load-1", reference: "DL-1001", origin: "Los Angeles, CA", destination: "Phoenix, AZ", rate: "$1200", status: "available" }
];

let fleet = [
  { id: "unit-1", name: "Truck 001", type: "Tractor", rate: "$500/day", status: "available" }
];

let driverLoads = {
  "driver-001": [
    { id: "load-1", reference: "DL-1001", origin: "Los Angeles, CA", destination: "Phoenix, AZ", status: "assigned" }
  ]
};

app.get('/health', (req, res) => {
  res.json({ status: "healthy" });
});

app.get('/loads', (req, res) => {
  res.json(loads);
});

app.get('/drivers/:driverId/loads', (req, res) => {
  const { driverId } = req.params;
  res.json(driverLoads[driverId] || []);
});

app.post('/loads/:loadId/status', (req, res) => {
  const { loadId } = req.params;
  const { status } = req.body;
  loads = loads.map(l => l.id === loadId ? { ...l, status } : l);
  Object.keys(driverLoads).forEach(did => {
    driverLoads[did] = (driverLoads[did] || []).map(l => l.id === loadId ? { ...l, status } : l);
  });
  res.json({ ok: true });
});

app.get('/fleet', (req, res) => {
  res.json(fleet);
});

const PORT = process.env.PORT || 4000;
app.listen(PORT, () => {
  console.log('Logistics API running on port', PORT);
});
```
